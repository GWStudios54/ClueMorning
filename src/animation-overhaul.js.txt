// Clue Morning Deep Cut canvas preview: single renderer, direct events, fail-open.
(() => {
  'use strict';

  const PREVIEW_PARAM = 'motion-preview';
  const PREVIEW_VALUE = 'deepcut';
  const ROUND_SECONDS = 25;
  const reducedMotion = window.matchMedia?.('(prefers-reduced-motion: reduce)');
  const nativeFetch = window.fetch?.bind(window);

  function previewEnabled() {
    try {
      return new URLSearchParams(window.location.search).get(PREVIEW_PARAM) === PREVIEW_VALUE;
    } catch {
      return false;
    }
  }

  function clamp(value, min, max) {
    return Math.max(min, Math.min(max, value));
  }

  function lerp(a, b, t) {
    return a + (b - a) * t;
  }

  function easeOutCubic(t) {
    return 1 - Math.pow(1 - clamp(t, 0, 1), 3);
  }

  function colorForTier(tier) {
    if (tier === 'rare') return '#e1aa45';
    if (tier === 'uncommon') return '#6fae89';
    return '#d8b36b';
  }

  function parseRound(text) {
    const match = String(text || '').match(/(\d+)\s*\/\s*(\d+)/);
    if (!match) return { shown: 1, total: 8 };
    return { shown: Number(match[1]) || 1, total: Number(match[2]) || 8 };
  }

  function isDeepCutCheck(input) {
    try {
      const raw = typeof input === 'string' ? input : input?.url;
      if (!raw) return false;
      const path = new URL(raw, window.location.href).pathname;
      return path === '/api/deepcut/check' || path === '/api/unlimited/deepcut/check';
    } catch {
      return false;
    }
  }

  function makeStage(panel) {
    const stage = document.createElement('div');
    stage.className = 'cm-deepcut-stage';
    stage.innerHTML = `
      <canvas class="cm-deepcut-canvas" aria-hidden="true"></canvas>
      <div class="cm-canvas-hud" aria-hidden="true">
        <span class="cm-depth-readout">ARCHIVE LEVEL <strong>01</strong></span>
        <span class="cm-scene-state">READY</span>
      </div>
    `;

    const stats = panel.querySelector('.stats-row');
    if (stats?.parentNode) stats.insertAdjacentElement('afterend', stage);
    else panel.prepend(stage);

    const back = document.createElement('button');
    back.type = 'button';
    back.className = 'cm-focus-exit';
    back.textContent = '← Clue Morning';
    back.addEventListener('click', () => {
      window.location.href = '/';
    });
    panel.prepend(back);

    return {
      stage,
      canvas: stage.querySelector('canvas'),
      depthReadout: stage.querySelector('.cm-depth-readout strong'),
      sceneState: stage.querySelector('.cm-scene-state')
    };
  }

  function createRenderer(panel, ui) {
    const canvas = ui.canvas;
    const ctx = canvas.getContext?.('2d', { alpha: false, desynchronized: true }) ||
      canvas.getContext?.('2d', { alpha: false });
    if (!ctx) throw new Error('Canvas 2D unavailable');

    const staticCanvas = document.createElement('canvas');
    const staticCtx = staticCanvas.getContext('2d', { alpha: false });
    if (!staticCtx) throw new Error('Canvas buffer unavailable');

    const roundInfo = parseRound(panel.querySelector('#deepCutRound')?.textContent);
    const status = panel.querySelector('#deepCutStatus')?.textContent?.trim() || 'READY';
    const initiallyDone = status === 'DONE';

    const state = {
      width: 0,
      height: 0,
      dpr: 1,
      raf: 0,
      resizeRaf: 0,
      lastTime: 0,
      active: status === 'LIVE',
      done: initiallyDone,
      urgent: false,
      deadline: 0,
      total: roundInfo.total,
      completed: initiallyDone ? roundInfo.total : Math.max(0, roundInfo.shown - 1),
      depth: initiallyDone ? roundInfo.total : Math.max(0, roundInfo.shown - 1),
      targetDepth: initiallyDone ? roundInfo.total : Math.max(0, roundInfo.shown - 1),
      descentFrom: 0,
      descentTo: 0,
      descentStart: 0,
      descentDuration: 780,
      shakeUntil: 0,
      shakePower: 0,
      flashUntil: 0,
      flashTier: 'common',
      completeStarted: initiallyDone ? performance.now() : 0,
      startPulseUntil: 0,
      timeoutHandled: false,
      quality: 1,
      frameWindow: [],
      reduced: Boolean(reducedMotion?.matches),
      motes: Array.from({ length: 18 }, (_, i) => ({
        x: ((i * 47) % 97) / 97,
        y: ((i * 31) % 89) / 89,
        r: 0.7 + (i % 4) * 0.35,
        speed: 7 + (i % 5) * 2.2,
        phase: i * 0.73
      }))
    };

    function cssVar(name, fallback) {
      const value = getComputedStyle(document.documentElement).getPropertyValue(name).trim();
      return value || fallback;
    }

    function drawStatic() {
      const { width: w, height: h, dpr } = state;
      staticCanvas.width = Math.max(1, Math.round(w * dpr));
      staticCanvas.height = Math.max(1, Math.round(h * dpr));
      staticCtx.setTransform(dpr, 0, 0, dpr, 0, 0);

      const paper = cssVar('--paper', '#f7f1e5');
      const dark = cssVar('--dark', '#423a31');
      const card = cssVar('--card', '#fffdf8');
      const sun = cssVar('--sun', '#d9ad62');

      staticCtx.fillStyle = dark;
      staticCtx.fillRect(0, 0, w, h);

      const wall = staticCtx.createLinearGradient(0, 0, 0, h);
      wall.addColorStop(0, '#40382f');
      wall.addColorStop(0.55, '#29251f');
      wall.addColorStop(1, '#1d1b18');
      staticCtx.fillStyle = wall;
      staticCtx.fillRect(0, 0, w, h);

      const centerW = Math.min(w * 0.66, 520);
      const left = (w - centerW) / 2;
      staticCtx.fillStyle = '#171613';
      staticCtx.fillRect(left, 0, centerW, h);

      staticCtx.fillStyle = '#514333';
      staticCtx.fillRect(left - 9, 0, 9, h);
      staticCtx.fillRect(left + centerW, 0, 9, h);

      staticCtx.fillStyle = '#806342';
      staticCtx.fillRect(left - 5, 0, 3, h);
      staticCtx.fillRect(left + centerW + 2, 0, 3, h);

      const beam = staticCtx.createLinearGradient(left, 0, left + centerW, 0);
      beam.addColorStop(0, 'rgba(0,0,0,.25)');
      beam.addColorStop(.45, 'rgba(235,195,119,.05)');
      beam.addColorStop(.55, 'rgba(255,225,165,.08)');
      beam.addColorStop(1, 'rgba(0,0,0,.3)');
      staticCtx.fillStyle = beam;
      staticCtx.fillRect(left, 0, centerW, h);

      staticCtx.fillStyle = paper;
      staticCtx.globalAlpha = .92;
      staticCtx.fillRect(left + 22, 18, Math.min(150, centerW * .42), 28);
      staticCtx.globalAlpha = 1;
      staticCtx.fillStyle = dark;
      staticCtx.font = '700 11px system-ui, sans-serif';
      staticCtx.fillText('DEEP ARCHIVE', left + 32, 36);

      staticCtx.fillStyle = sun;
      staticCtx.fillRect(left + centerW - 62, 23, 32, 4);
      staticCtx.fillStyle = card;
      staticCtx.globalAlpha = .18;
      staticCtx.fillRect(left + 18, h - 22, centerW - 36, 2);
      staticCtx.globalAlpha = 1;
    }

    function resize() {
      const rect = ui.stage.getBoundingClientRect();
      const width = Math.max(280, Math.round(rect.width));
      const height = Math.max(240, Math.round(rect.height));
      const mobile = window.matchMedia?.('(max-width: 760px)').matches;
      const dprCap = mobile ? 1.35 : 1.75;
      const dpr = clamp(window.devicePixelRatio || 1, 1, dprCap);

      if (width === state.width && height === state.height && dpr === state.dpr) return;
      state.width = width;
      state.height = height;
      state.dpr = dpr;
      canvas.width = Math.max(1, Math.round(width * dpr));
      canvas.height = Math.max(1, Math.round(height * dpr));
      canvas.style.width = `${width}px`;
      canvas.style.height = `${height}px`;
      drawStatic();
      draw(performance.now());
    }

    function drawShelf(y, level, t) {
      const w = state.width;
      const centerW = Math.min(w * 0.66, 520);
      const left = (w - centerW) / 2;
      const right = left + centerW;
      const hue = level % 3;
      const wood = hue === 0 ? '#6e5238' : hue === 1 ? '#765a3b' : '#654a34';
      const paper = hue === 0 ? '#d9cbb3' : hue === 1 ? '#c8b795' : '#e0d3bd';

      ctx.fillStyle = wood;
      ctx.fillRect(left + 14, y, centerW - 28, 7);

      const shelfH = 60;
      for (const side of [-1, 1]) {
        const x0 = side < 0 ? left + 18 : right - 104;
        ctx.fillStyle = 'rgba(13,12,10,.55)';
        ctx.fillRect(x0, y - shelfH, 86, shelfH);

        for (let i = 0; i < 6; i++) {
          const bh = 18 + ((i * 13 + level * 7) % 31);
          const bw = 6 + ((i + level) % 3) * 2;
          const x = x0 + 6 + i * 12;
          ctx.fillStyle = i % 2 ? paper : '#aa936e';
          ctx.globalAlpha = .7;
          ctx.fillRect(x, y - 6 - bh, bw, bh);
        }
        ctx.globalAlpha = 1;
      }

      ctx.fillStyle = 'rgba(229,190,115,.78)';
      ctx.fillRect(left + centerW * .5 - 1, y - 42, 2, 22);
      ctx.fillStyle = 'rgba(244,230,200,.72)';
      ctx.font = '700 10px system-ui, sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText(String(level + 1).padStart(2, '0'), left + centerW * .5, y - 49);

      if (t > 0) {
        ctx.globalAlpha = Math.min(.28, t * .28);
        ctx.fillStyle = '#e1aa45';
        ctx.fillRect(left + 22, y - shelfH, centerW - 44, 2);
        ctx.globalAlpha = 1;
      }
    }

    function drawLift(now) {
      const w = state.width;
      const h = state.height;
      const centerW = Math.min(w * 0.66, 520);
      const x = w / 2;
      const baseY = h * .57;
      let shakeX = 0;
      let shakeY = 0;

      if (now < state.shakeUntil) {
        const left = (state.shakeUntil - now) / 360;
        const p = state.shakePower * left;
        shakeX = Math.sin(now * .075) * p;
        shakeY = Math.cos(now * .11) * p * .35;
      }

      const pulse = now < state.startPulseUntil
        ? Math.sin((state.startPulseUntil - now) * .018) * .5 + .5
        : 0;

      const completeT = state.completeStarted
        ? clamp((now - state.completeStarted) / 1200, 0, 1)
        : 0;

      const y = baseY + shakeY - completeT * h * .08;
      const liftW = Math.min(176, centerW * .48);
      const liftH = 92;

      ctx.save();
      ctx.translate(shakeX, 0);

      ctx.fillStyle = '#b88843';
      ctx.fillRect(x - liftW / 2 - 5, y - liftH / 2 - 5, liftW + 10, liftH + 10);
      ctx.fillStyle = '#342d26';
      ctx.fillRect(x - liftW / 2, y - liftH / 2, liftW, liftH);

      ctx.fillStyle = '#151514';
      ctx.fillRect(x - liftW * .38, y - liftH * .34, liftW * .76, liftH * .56);

      const glowAlpha = state.urgent ? .18 + (Math.sin(now * .012) + 1) * .08 : .08 + pulse * .16;
      ctx.fillStyle = state.urgent
        ? `rgba(180,72,58,${glowAlpha})`
        : `rgba(228,180,93,${glowAlpha})`;
      ctx.fillRect(x - liftW * .38, y - liftH * .34, liftW * .76, liftH * .56);

      for (const lx of [x - liftW * .3, x + liftW * .3]) {
        ctx.fillStyle = state.urgent ? '#cf6757' : '#e1aa45';
        ctx.beginPath();
        ctx.arc(lx, y + liftH * .31, 4, 0, Math.PI * 2);
        ctx.fill();
      }

      ctx.fillStyle = '#d8ccb6';
      ctx.fillRect(x - 35, y + liftH * .12, 70, 13);
      ctx.fillStyle = '#6a5c49';
      ctx.fillRect(x - 26, y + liftH * .17, 52, 2);

      ctx.restore();

      if (completeT > 0) {
        ctx.globalAlpha = completeT * .82;
        ctx.fillStyle = '#e8c986';
        ctx.fillRect(0, 0, w, h);
        ctx.globalAlpha = 1;
      }
    }

    function drawMotes(now) {
      const max = Math.max(6, Math.round(state.motes.length * state.quality));
      ctx.fillStyle = state.urgent ? 'rgba(219,122,102,.42)' : 'rgba(238,211,156,.34)';
      for (let i = 0; i < max; i++) {
        const mote = state.motes[i];
        const y = ((mote.y * state.height + now * mote.speed * .001) % (state.height + 24)) - 12;
        const x = mote.x * state.width + Math.sin(now * .0012 + mote.phase) * 8;
        ctx.beginPath();
        ctx.arc(x, y, mote.r, 0, Math.PI * 2);
        ctx.fill();
      }
    }

    function drawFlash(now) {
      if (now >= state.flashUntil) return;
      const remaining = (state.flashUntil - now) / 720;
      const alpha = Math.sin(clamp(remaining, 0, 1) * Math.PI) * .22;
      ctx.globalAlpha = alpha;
      ctx.fillStyle = colorForTier(state.flashTier);
      ctx.fillRect(0, 0, state.width, state.height);
      ctx.globalAlpha = 1;

      const count = state.flashTier === 'rare' ? 12 : state.flashTier === 'uncommon' ? 9 : 7;
      const progress = 1 - clamp(remaining, 0, 1);
      const radius = 24 + easeOutCubic(progress) * 88;
      ctx.save();
      ctx.translate(state.width / 2, state.height * .48);
      ctx.fillStyle = colorForTier(state.flashTier);
      for (let i = 0; i < count; i++) {
        const a = (Math.PI * 2 * i) / count + .13;
        const x = Math.cos(a) * radius;
        const y = Math.sin(a) * radius * .62;
        ctx.globalAlpha = (1 - progress) * .9;
        ctx.fillRect(x - 3, y - 7, 6, 14);
      }
      ctx.restore();
      ctx.globalAlpha = 1;
    }

    function updateDepth(now) {
      if (!state.descentStart) return;
      const progress = clamp((now - state.descentStart) / state.descentDuration, 0, 1);
      state.depth = lerp(state.descentFrom, state.descentTo, easeOutCubic(progress));
      if (progress >= 1) {
        state.depth = state.descentTo;
        state.descentStart = 0;
      }
    }

    function updateTimer() {
      if (!state.active || !state.deadline || state.done) {
        state.urgent = false;
        panel.classList.remove('cm-canvas-urgent');
        return;
      }

      const remaining = (state.deadline - Date.now()) / 1000;
      const urgent = remaining <= 5 && remaining > 0;
      if (urgent !== state.urgent) {
        state.urgent = urgent;
        panel.classList.toggle('cm-canvas-urgent', urgent);
        if (urgent) emit('urgent', { remaining });
      }

      if (remaining <= 0 && !state.timeoutHandled) {
        state.timeoutHandled = true;
        emit('timeout');
        if (!state.done) {
          state.deadline = Date.now() + ROUND_SECONDS * 1000;
          state.timeoutHandled = false;
        }
      }
    }

    function draw(now) {
      if (!state.width || !state.height) return;
      updateDepth(now);
      updateTimer(now);

      ctx.setTransform(state.dpr, 0, 0, state.dpr, 0, 0);
      ctx.clearRect(0, 0, state.width, state.height);
      ctx.drawImage(staticCanvas, 0, 0, state.width, state.height);

      const shelfSpacing = 78;
      const scroll = (state.depth * shelfSpacing) % shelfSpacing;
      const baseLevel = Math.floor(state.depth);
      const shelfCount = Math.ceil(state.height / shelfSpacing) + 3;
      for (let i = -1; i < shelfCount; i++) {
        const y = i * shelfSpacing - scroll + 86;
        const level = Math.max(0, baseLevel + i);
        drawShelf(y, level, state.descentStart ? .7 : 0);
      }

      drawMotes(now);
      drawLift(now);
      drawFlash(now);

      if (!canvas.dataset.rendered) {
        canvas.dataset.rendered = 'true';
        panel.dataset.canvasRendered = 'true';
      }
    }

    function frame(now) {
      state.raf = 0;
      if (document.visibilityState === 'hidden' || !panel.classList.contains('active')) return;

      if (state.lastTime) {
        const dt = now - state.lastTime;
        state.frameWindow.push(dt);
        if (state.frameWindow.length > 45) state.frameWindow.shift();
        if (state.frameWindow.length === 45) {
          const avg = state.frameWindow.reduce((a, b) => a + b, 0) / state.frameWindow.length;
          state.quality = avg > 24 ? .58 : 1;
        }
      }
      state.lastTime = now;
      draw(now);

      if (!state.reduced) state.raf = requestAnimationFrame(frame);
    }

    function startLoop() {
      if (state.reduced) {
        draw(performance.now());
        return;
      }
      if (!state.raf && document.visibilityState !== 'hidden' && panel.classList.contains('active')) {
        state.lastTime = 0;
        state.raf = requestAnimationFrame(frame);
      }
    }

    function stopLoop() {
      if (state.raf) cancelAnimationFrame(state.raf);
      state.raf = 0;
      state.lastTime = 0;
    }

    function setDepth(nextDepth) {
      const target = clamp(nextDepth, 0, state.total);
      state.descentFrom = state.depth;
      state.descentTo = target;
      state.targetDepth = target;
      state.descentStart = performance.now();
      ui.depthReadout.textContent = String(Math.min(state.total, Math.floor(target) + 1)).padStart(2, '0');
    }

    function emit(eventName, detail = {}) {
      const now = performance.now();
      panel.dataset.lastMotionEvent = eventName;

      if (eventName === 'start') {
        state.active = true;
        state.done = false;
        state.deadline = Date.now() + ROUND_SECONDS * 1000;
        state.timeoutHandled = false;
        state.startPulseUntil = now + 900;
        panel.dataset.motionState = 'live';
        ui.sceneState.textContent = 'DESCENDING';
      } else if (eventName === 'wrong') {
        state.shakeUntil = now + 360;
        state.shakePower = 8;
        ui.sceneState.textContent = 'KEEP DIGGING';
      } else if (eventName === 'correct_common' || eventName === 'correct_uncommon' || eventName === 'correct_rare') {
        const tier = eventName.replace('correct_', '');
        state.flashTier = tier;
        state.flashUntil = now + (tier === 'rare' ? 900 : 720);
        state.completed = clamp(state.completed + 1, 0, state.total);
        state.deadline = Date.now() + ROUND_SECONDS * 1000;
        state.timeoutHandled = false;
        setDepth(state.completed);
        panel.dataset.motionState = tier;
        ui.sceneState.textContent = tier.toUpperCase();
        if (state.completed >= state.total) {
          window.setTimeout(() => emit('complete'), 620);
        }
      } else if (eventName === 'timeout') {
        state.shakeUntil = now + 420;
        state.shakePower = 5;
        state.completed = clamp(state.completed + 1, 0, state.total);
        setDepth(state.completed);
        panel.dataset.motionState = 'timeout';
        ui.sceneState.textContent = 'TIME';
        if (state.completed >= state.total) {
          window.setTimeout(() => emit('complete'), 520);
        }
      } else if (eventName === 'urgent') {
        panel.dataset.motionState = 'urgent';
        ui.sceneState.textContent = '5 SECONDS';
      } else if (eventName === 'complete') {
        state.active = false;
        state.done = true;
        state.deadline = 0;
        state.urgent = false;
        state.completeStarted = now;
        panel.classList.remove('cm-canvas-urgent');
        panel.classList.add('cm-canvas-complete');
        panel.dataset.motionState = 'complete';
        ui.sceneState.textContent = 'ARCHIVE COMPLETE';
      }

      if (state.reduced) draw(now);
      startLoop();
    }

    function onResize() {
      if (state.resizeRaf) return;
      state.resizeRaf = requestAnimationFrame(() => {
        state.resizeRaf = 0;
        resize();
      });
    }

    window.addEventListener('resize', onResize, { passive: true });
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden') stopLoop();
      else {
        resize();
        startLoop();
      }
    }, { passive: true });

    resize();
    startLoop();

    return { state, emit, resize, startLoop, stopLoop };
  }

  function installDeepCut() {
    const panel = document.querySelector('#deepcut');
    const startButton = document.querySelector('#deepCutStart');
    const form = document.querySelector('#deepCutForm');
    if (!panel || !form) return null;
    if (panel.dataset.motionInstalled === 'true') return window.ClueMotion?.renderer || null;

    document.body.classList.add('cm-deepcut-focus');
    panel.classList.add('cm-deepcut-canvas-page');
    panel.dataset.motionInstalled = 'true';
    panel.dataset.motionState = 'ready';

    if (!panel.classList.contains('active')) {
      document.querySelector('[data-tab="deepcut"]')?.click();
    }

    const ui = makeStage(panel);
    const renderer = createRenderer(panel, ui);

    startButton?.addEventListener('click', () => renderer.emit('start'), { capture: true });

    form.addEventListener('submit', () => {
      if (!renderer.state.active || renderer.state.done) return;
      ui.sceneState.textContent = 'CHECKING';
    }, { capture: true });

    if (nativeFetch) {
      window.fetch = async (...args) => {
        const response = await nativeFetch(...args);
        if (!isDeepCutCheck(args[0])) return response;

        response.clone().json().then(result => {
          window.setTimeout(() => {
            if (!result?.accepted) {
              renderer.emit('wrong');
              return;
            }
            const tier = String(result.tier || 'COMMON').toLowerCase();
            const safeTier = tier === 'rare' ? 'rare' : tier === 'uncommon' ? 'uncommon' : 'common';
            renderer.emit(`correct_${safeTier}`, {
              answer: result.canonical || '',
              score: Number(result.score || 0)
            });
          }, 0);
        }).catch(() => {
          // The base game owns error handling. Animation remains fail-open.
        });
        return response;
      };
    }

    const status = panel.querySelector('#deepCutStatus')?.textContent?.trim();
    if (status === 'LIVE') renderer.emit('start');
    if (status === 'DONE') renderer.emit('complete');

    return renderer;
  }

  const publicApi = window.ClueMotion || {};
  publicApi.preview = PREVIEW_VALUE;
  publicApi.engine = 'canvas2d';
  publicApi.emit = (eventName, detail) => publicApi.renderer?.emit?.(eventName, detail);
  window.ClueMotion = publicApi;

  function boot() {
    if (!previewEnabled()) return;
    try {
      publicApi.renderer = installDeepCut();
      window.dispatchEvent(new CustomEvent('clue-motion-ready', {
        detail: { preview: PREVIEW_VALUE, engine: 'canvas2d' }
      }));
    } catch (error) {
      console.warn('Clue Morning canvas preview disabled after a safe failure.', error);
    }
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot, { once: true });
  else boot();
})();
